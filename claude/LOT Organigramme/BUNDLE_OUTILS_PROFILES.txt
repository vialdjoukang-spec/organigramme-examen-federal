"""Contrôle de conformité V5 d'un fichier de lot. Usage: python3 v5_lint.py fichier.html [n_monographies]"""
import re, sys
from html.parser import HTMLParser

class P(HTMLParser):
    def __init__(s): super().__init__(); s.stack=[]; s.err=[]; s.text=[]
    VOID={'br','meta','img','hr','input','link'}
    def handle_starttag(s,t,a):
        if t not in s.VOID: s.stack.append(t)
        if t=='a' and dict(a).get('href')=='#': s.err.append('lien href="#" interdit')
    def handle_endtag(s,t):
        if t in s.VOID: return
        if t in s.stack:
            while s.stack and s.stack[-1]!=t: s.err.append(f'balise non fermée <{s.stack.pop()}>')
            s.stack.pop()
        else: s.err.append(f'fermeture orpheline </{t}>')
    def handle_data(s,d): s.text.append(d)

f=sys.argv[1]; n=int(sys.argv[2]) if len(sys.argv)>2 else None
h=open(f,encoding='utf-8').read(); p=P(); p.feed(h); err=list(p.err)
if p.stack: err.append(f'non fermés en fin: {p.stack}')
for need in ('<!DOCTYPE html>','charset="UTF-8"','name="viewport"'):
    if need not in h: err.append(f'manque {need}')
mono=len(re.findall(r'<details open',h))
if n and mono!=n: err.append(f'{mono} monographies au lieu de {n}')
for k in range(1,15):
    c=len(re.findall(rf'<strong>{k}\.\s',h))
    if n and c<n and h.count('plan-academique')==0: err.append(f'point {k}: {c}/{n}')
for sub in ('A- But','B- Moyens','C- Surveillance','D- Pronostic'):
    c=h.count(sub)
    if n and c<n and h.count('plan-academique')==0: err.append(f'11.{sub}: {c}/{n}')
if 'title="' in re.sub(r'<title>.*?</title>','',h,flags=re.S) and '<a ' in h: err.append('infobulle title= détectée')
if 'FIN_FICHIER' not in h: err.append('balise FIN_FICHIER absente')
if 'non validé cliniquement' not in h: err.append('mention de non-validation absente')
if not re.search(r'Sources',h): err.append('section Sources absente')
txt=' '.join(p.text)
pct=[m for m in re.findall(r'[^.]{0,60}\d+(?:[.,]\d+)?\s?%',txt) if not re.search(r'\[S\d+|S\d+\]|Source|source',m)]
print(f'{f}: {mono} monographies, {len(h)//1024} Ko, {len(err)} erreur(s)')
for e in err[:40]: print('  ERR', e)
print(f'  info: {len(pct)} pourcentages à vérifier contre evidence.json')
sys.exit(1 if err else 0)

===== FICHIER: wrap_lines.py =====
"""Coupe les lignes HTML trop longues après les balises de bloc (rendu inchangé)."""
import re,sys
for f in sys.argv[1:]:
    h=open(f,encoding='utf-8').read()
    out=[]
    for line in h.split('\n'):
        if len(line)>1500: line=re.sub(r'(</(?:p|div|details|summary|tr|li|ul|ol|table|thead|tbody|h[1-6])>)',r'\1\n',line)
        out.append(line)
    n='\n'.join(out); open(f,'w',encoding='utf-8').write(n)
    print(f.split('/')[-1][:40], max(len(l) for l in n.split('\n')))

===== FICHIER: facts_check.py =====
"""Vérifie qu'une réécriture conserve les faits : nombres, sources [Sn], spécialités ®. Usage: facts_check.py avant.html apres.html"""
import re,sys,html
def facts(p):
    t=html.unescape(re.sub(r'<[^>]+>',' ',open(p,encoding='utf-8').read()))
    nums=set(re.findall(r'\d+(?:[.,]\d+)?',t)); src=set(re.findall(r'\bS\d+\b',t)); brands=set(re.findall(r'[\w-]+®',t))
    return nums,src,brands
a,b=facts(sys.argv[1]),facts(sys.argv[2])
lost=[sorted(x-y) for x,y in zip(a,b)]
print('nombres perdus:',lost[0][:60]); print('sources perdues:',lost[1]); print('spécialités perdues:',lost[2])
sys.exit(1 if any(lost) else 0)

"""Rendu réel : python3 render_check.py fichier.html -> largeur 390 px, débordement horizontal, ouverture des <details>, capture."""
import sys, pathlib, glob
from playwright.sync_api import sync_playwright
f=pathlib.Path(sys.argv[1]).resolve()
exe=next(iter(glob.glob('/opt/pw-browsers/chromium-*/chrome-linux*/chrome')),None)
with sync_playwright() as p:
    b=p.chromium.launch(**({'executable_path':exe} if exe else {}))
    pg=b.new_page(viewport={'width':390,'height':844}); errs=[]
    pg.on('pageerror',lambda e: errs.append(str(e)))
    pg.goto(f.as_uri())
    over=pg.evaluate("document.documentElement.scrollWidth-window.innerWidth")
    n=pg.evaluate("document.querySelectorAll('details').length")
    pg.evaluate("document.querySelectorAll('details').forEach(d=>d.open=true)")
    wide=pg.evaluate("[...document.querySelectorAll('body *')].filter(e=>e.getBoundingClientRect().right>window.innerWidth+2 && !e.closest('[style*=overflow]')).length")
    pg.screenshot(path=str(f.with_suffix('.mobile.png')))
    print(f'{f.name}: details={n} debordement_page={over}px elements_hors_cadre={wide} erreurs_js={len(errs)}')
    b.close()

## Consigne de Vial (2026-10-21) — plan académique de repli et qualité des différentiels
11. **Plan de repli.** Quand le plan monographique en 14 points ne s'applique pas naturellement à une entité (SSP transversal, procédural, médico-légal, ou entité qui est une classe d'agents plutôt qu'une maladie), ne pas forcer : produire un **plan universitaire/académique** (I., II., III. ; A., B. ; 1., 2.) qui suit la logique propre du sujet, avec **de nombreuses fenêtres cliquables `<details>`** pour les détails (mécanismes, calculs, posologies, cadre légal, exemples). Annoncer en tête de la monographie, en une phrase, pourquoi ce plan est retenu. Les exigences de fond (sources, posologies suisses, bio-mathématique utile, guérison/recherche 2026, pièges d'examen) restent dues ; le lint accepte ce plan si la monographie contient la classe `plan-academique`.
12. **Différentiels de haut niveau.** Chaque DX doit dire ce qui le **distingue des autres DX** et des simulateurs non toxiques :
   - Appel A, Bloc 2 : outre la liste ordonnée, un **tableau comparatif des toxidromes/entités** (conscience, pupilles, FC/PA, FR, température, peau, péristaltisme, ECG, biologie clé, antidote) et un **algorithme discriminant** (quel signe ou examen fait basculer d'un DX à l'autre).
   - Chaque monographie Appel B : rubrique « Diagnostics différentiels discriminants » au point 4 (ou section équivalente), avec les 3–4 confusions les plus dangereuses, le signe ou examen qui tranche, et le piège d'examen associé ; inclure les causes non toxiques qui miment l'intoxication (hypoglycémie, AVC, méningo-encéphalite, état de mal non convulsif, sepsis, hypothermie, traumatisme crânien, troubles métaboliques).
   - Associations et co-ingestions (alcool + benzodiazépines + paracétamol) explicitement traitées : toujours doser paracétamol et éthanol dans l'ingestion volontaire inconnue si la source le soutient.
13. **Fenêtres des thèmes transversaux : très riches et très précises.** Pour tout thème transversal (cadre légal, éthique, communication, organisation des soins, prévention, santé publique, évaluation du risque suicidaire, capacité de discernement, déclaration, Tox Info Suisse, médecine légale), chaque fenêtre `<details>` contient : définition exacte ; base légale ou déontologique précise (article et alinéa via Fedlex, directive ASSM ou code FMH avec année) ; conditions d'application et exceptions ; procédure pas à pas ; exemple clinique concret avec la décision correcte ; erreur fréquente ou piège d'examen ; source datée. Une fenêtre transversale ne se limite jamais à une phrase.
